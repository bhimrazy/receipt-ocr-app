import { randomUUID } from "node:crypto";
import { sql, supportMessages } from "@receipt-ocr/db";
import { Cause, Effect, Exit, Layer, ManagedRuntime } from "effect";
import { afterAll, expect, it } from "vitest";
import { makeAppLayer } from "../src/index";
import { ContactReplyMailer, SupportNotifyMailer, WelcomeMailer } from "../src/jobs";
import { Db } from "../src/services";

const runtime = ManagedRuntime.make(
  makeAppLayer(
    Layer.mergeAll(WelcomeMailer.Disabled, ContactReplyMailer.Disabled, SupportNotifyMailer.Disabled),
  ),
);
const tag = `zzrepro-${Date.now()}`;
// Same order as seed(): closedAt computed by the caller, createdAt inside.
const insert = (closedAt: Date) => {
  return Effect.flatMap(Db, ({ use }) =>
    use((d) =>
      d.insert(supportMessages).values({
        submissionId: randomUUID(),
        kind: "question",
        source: "public",
        status: "closed",
        name: "T",
        email: `${tag}-${randomUUID().slice(0, 8)}@example.com`,
        message: "A test question",
        createdAt: new Date(),
        updatedAt: closedAt,
        closedAt,
      }),
    ),
  );
};
afterAll(async () => {
  await runtime.runPromise(
    Effect.flatMap(Db, ({ use }) =>
      use((d) => d.delete(supportMessages).where(sql`${supportMessages.email} like ${`${tag}%`}`)),
    ),
  );
  await runtime.dispose();
});
it("seed pattern", async () => {
  let fails = 0;
  let first: unknown;
  const N = Number(process.env.N ?? 3000);
  for (let i = 0; i < N; i++) {
    const exit = await runtime.runPromiseExit(insert(new Date()));
    if (Exit.isFailure(exit)) {
      fails++;
      first ??= Cause.squash(exit.cause);
    }
  }
  const e = first as
    | { cause?: { message?: string; cause?: { message?: string; code?: string; constraint?: string } } }
    | undefined;
  process.stdout.write(
    `REPRO ${JSON.stringify({ N, fails, pg: e?.cause?.cause?.message, code: e?.cause?.cause?.code, constraint: e?.cause?.cause?.constraint, drizzle: e?.cause?.message?.slice(0, 60) })}\n`,
  );
  expect(true).toBe(true);
}, 600_000);
